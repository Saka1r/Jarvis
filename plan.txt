jarvis/
	- init.py
	- main.py Запускает Jarvis
	- config.py конфигурация. Настройка (пути, порты, токены и т.д). Сборка всего необходимого для проекта с data и т.д.
	
	config/
		
		- system.json -> настройки системы

	core/
		- init.py
		- jarvis.py ядро проекта
		- registry.py загружает плагины 
		- context.py хранение состояния/сессий 

	subsystems/
		- init.py
		- audio_in.py (микрофон/входные данные)
		- audio_out.py (TTS, воспроизведение)
		- stt.py (STT адаптеры)
			+ vosk_stt.py
		- tts.py (TTS)

	plugins/
		...  плагины

	storage/
		- init.py
		- file_storage.py (json)
		# не точно
	data/

		- settings.json
		- plugins.json

		... -> файлы необходимые для Джарвиса по типу референсы голоса, модели и т.д

		recordings/
			... -> временные записи

	utils/
		- init.py
		- log.py
	
	tests/

    "version": "1.0",

    "language": "ru"
  },

  "ui": {

Насчет плагинов

project/
├── plugins.json                 # Список установленных плагинов
├── settings.json                # Настройки приложения
├── voice_commands_cache.json    # (опционально) кэш для быстрого доступа
└── plugins/
    ├── jarvis/
    │   ├── plugin.json
    │   ├── commands.json        # Команды плагина
    │   └── main.py
    └── assistant/
        ├── plugin.json
        ├── commands.json
        └── main.py
